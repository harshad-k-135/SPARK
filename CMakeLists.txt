cmake_minimum_required(VERSION 3.24)
project(SPARK VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "Target CUDA compute capabilities")

add_library(spark_build_options INTERFACE)
if(MSVC)
    target_compile_options(spark_build_options INTERFACE $<$<CONFIG:Release>:/O2> /utf-8)
else()
    target_compile_options(spark_build_options INTERFACE $<$<CONFIG:Release>:-O3>)
endif()

add_library(spark_model engine/model_loader/gguf.cpp engine/model_loader/packed_weights.cpp)
target_include_directories(spark_model PUBLIC ${PROJECT_SOURCE_DIR})
target_link_libraries(spark_model PUBLIC spark_build_options)

add_library(spark_sampling engine/sampling/sampler.cpp)
target_include_directories(spark_sampling PUBLIC ${PROJECT_SOURCE_DIR})
target_link_libraries(spark_sampling PUBLIC spark_build_options)

add_library(spark_qwen_ops engine/transformer/delta_net.cpp)
target_include_directories(spark_qwen_ops PUBLIC ${PROJECT_SOURCE_DIR})
target_link_libraries(spark_qwen_ops PUBLIC spark_build_options)

add_library(spark_cpu engine/transformer/cpu_model.cpp)
target_link_libraries(spark_cpu PUBLIC spark_qwen_ops spark_model spark_sampling)
find_package(OpenMP QUIET COMPONENTS CXX)
if(OpenMP_CXX_FOUND)
    target_link_libraries(spark_cpu PRIVATE OpenMP::OpenMP_CXX)
endif()

add_executable(spark engine/runtime/cli.cpp)
add_library(spark_tokenizer engine/tokenizer/tokenizer.cpp)
target_link_libraries(spark_tokenizer PUBLIC spark_model)
target_link_libraries(spark PRIVATE spark_tokenizer spark_cpu)
find_package(CUDAToolkit 12 QUIET)
if(CUDAToolkit_FOUND)
    add_library(spark_cuda cuda/qwen_gpu.cpp cuda/memory/arena.cpp)
    target_compile_definitions(spark_cuda PRIVATE SPARK_COMPUTE_ARCH="${CMAKE_CUDA_ARCHITECTURES}")
    target_link_libraries(spark_cuda PRIVATE spark_model spark_qwen_ops CUDA::cuda_driver CUDA::cudart CUDA::nvrtc CUDA::cublas)
    if(OpenMP_CXX_FOUND)
        target_link_libraries(spark_cuda PRIVATE OpenMP::OpenMP_CXX)
    endif()
    target_link_libraries(spark PRIVATE spark_cuda)
    target_compile_definitions(spark PRIVATE SPARK_WITH_CUDA)
endif()
if(MINGW)
    target_link_options(spark PRIVATE -municode)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(spark_packed_weights_test tests/unit/packed_weights.cpp)
    target_link_libraries(spark_packed_weights_test PRIVATE spark_model)
    add_test(NAME spark_packed_weights COMMAND spark_packed_weights_test)
    add_executable(spark_delta_net_test tests/unit/delta_net.cpp)
    target_link_libraries(spark_delta_net_test PRIVATE spark_qwen_ops)
    add_test(NAME spark_delta_net COMMAND spark_delta_net_test)
    add_executable(spark_sampler_test tests/unit/sampler.cpp)
    target_link_libraries(spark_sampler_test PRIVATE spark_sampling)
    add_test(NAME spark_sampler COMMAND spark_sampler_test)
    if(CUDAToolkit_FOUND)
        add_executable(spark_cuda_test tests/unit/qwen_cuda.cpp)
        target_link_libraries(spark_cuda_test PRIVATE spark_cuda spark_model)
        add_test(NAME spark_cuda_ops COMMAND spark_cuda_test)
    endif()
endif()
